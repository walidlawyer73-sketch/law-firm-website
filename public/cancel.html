document.addEventListener('DOMContentLoaded', function() { const langSwitch = document.querySelector('.language-switcher a'); if (langSwitch) { const currentPath = window.location.pathname; // إذا كان المسار يبدأ بـ /ar، نذهب للإنجليزية، وإلا نذهب للعربية if (currentPath.startsWith('/ar/')) { // الصفحة الحالية عربية، نذهب للإنجليزية const englishPath = currentPath.replace('/ar/', '/'); langSwitch.href = englishPath || '/'; langSwitch.classList.remove('arabic'); langSwitch.innerHTML = ' English'; } else { // الصفحة الحالية إنجليزية، نذهب للعربية const arabicPath = currentPath === '/' ? '/ar/' : `/ar${currentPath}`; langSwitch.href = arabicPath; langSwitch.classList.add('arabic'); langSwitch.innerHTML = ' العربية'; } } });
